## 零、本轮做了什么

沿「压力 → 判据 → 采集 → 打包 → PWM → 时序」六层逐层追反汇编，**把 TF100A 的压力/触觉链路追到底**，
并定位了两个此前从未识别的关键立即数：

| 立即数 | 地址 | 含义 |
|---|---|---|
| **`0x28`（40）** | **`0x08005A90` / `0x08005ABC`** | ★★★ **突变检测灵敏度**（判据的分母） |
| **`0x2BC`（700）** | `0x080059F0` | 输入计数的量程门（≤700 才继续） |

**★ 以及一个否定性但重要的发现：运行期**从不修改 CCR**（占空比固定），PWM 在上电初始化时就被启动。**

## 一、完整震动逻辑（六层）

### 第 1 层 · 压力采集 —— **TF100A 自己的 ADC**

```asm
08005CD4  push {r7, lr}                  ← 采集初始化函数
08005CE4  r0 = 0x40010800                ← GPIOA
08005CF6  bl 0x0800F81C                  ← 配引脚（模拟输入）
08005CFE  r0 = 0x40010C00                ← GPIOB
08005D0A  bl 0x0800F81C
08005D10  r0 = 0x40012400                ← ★★★ ADC1
08005D2C  bl 0x0800EF20                  ← ADC 配置
08005D50  bl 0x0800EDAC
```

**★ 地址映射与 STM32F1 完全吻合：`GPIOA=0x40010800` · `GPIOB=0x40010C00` · `ADC1=0x40012400` · `I2C1=0x40005400`（见 `0x0800894E`）· `TIM2=0x40000000` · `TIM3=0x40000400`**

> ## **⇒ 4 个压力通道 = TF100A 自己 ADC1 的 4 路模拟输入（4 个力传感器），不是从 GT7868Q 传来的。**
> **⇒ 因此压力阈值确实在 TF100A 固件里 ⇒ 改它有实际意义。**

### 第 2 层 · 累加求平均（降噪）

```asm
08005BB8  push {r7, lr}                  ← 采样任务
08005BC2  r0 = [ctx + 0x1691]            读采样窗口标志
08005BC8  cmp r0, #1 ; bne → 返回        窗口未开则退出
08005BE0  and r0, #0xfe ; strb           清标志
08005BEE  i = 0 ; cmp i, #3 ; bgt         循环 4 通道
08005C02    r3 = [源 + i*2 + 2]           ★ 第 i 通道原始值（16 位）
08005C0C    r0 = [0x2000405C + i*4]       4 个 32 位累加器
08005C12    [0x2000405C + i*4] += r3
08005C24  [0x20004058]++                 采样计数
08005C34  cmp.w r0, #0x12c (300)         ★ 满 300 次
08005C3C    [ctx+0x1691] = 0             关窗口
08005C4C    [ctx+0x1692] = 1             标记「累积完成」
08005C64    r0 = [0x2000405C + i*4]      取累加值
08005C70    r1 = 0x1B4E81B5              ★★★ 除法魔数
08005C78    umull
08005C7C    r0 >>= 5
08005C84    strh r0, [结构 + 0xa + i*2]  ★ 存回 16 位平均值
08005C92    r0 = 0x08011A40
08005C9A    bl 0x080110BC                ★ 派发（通用上报包装）
```

**★ 魔数验证：`2^(32+5) / 300 = 2^37/300 = 458,129,844.9 ≈ 0x1B4E81B4`（代码 `0x1B4E81B5`）**
> **⇒ 除数是 300 ⇒ `累加值 ÷ 300 = 平均压力`，存成 16 位，4 通道各自独立。**

### 第 3 层 · ★★★ 判据：`|基线 − 当前| > 40`

```asm
080059E0  push {r7, lr}                  ← 压力处理函数
080059E4  str r0, [sp, #0xc]             参数1 = 输入值数组
080059E6  str r1, [sp, #8]               参数2 = ctx（0x2000495E）
080059E8  strh r2, [sp, #6]              ★ 参数3 = 16 位计数值
080059EC  r0 = [sp, #6]
080059F0  cmp.w r0, #0x2bc (700)         ★ 量程门
080059F4  ble → 继续                      ≤700 才处理
080059F8  b   → 0x08005BB4                >700 直接返回

; —— 检查「处理中」标志，避免重入 ——
080059FC  r0 = [ctx + 0x168a]
08005A02  cmp r0, #0
08005A04  bne → 0x08005B12

; —— 阶段1：4 通道原始值写入历史环形缓冲 ——
08005A0A  bl 0x0800B760
08005A16  i = 0 … cmp #3
08005A30    r0 = [0x2000403C + i*2 + 2]  取原始值
08005A3C    r3 = [ctx + 0x1688]          当前环形索引
08005A3C    dst = ctx + i*0x28 + [ctx+0x1688]*2 + 0x15E8
08005A4A    strh r0, [dst]

; —— 阶段2：逐通道判据 ——
08005A5A  j = 0 … cmp #3
08005A88    r0 = 基线值（从历史缓冲取）
08005A8A    r1 = [参数1][j*2]            当前值
08005A90    adds r1, #0x28   (40)        ★★★ 基线 > 当前+40 ?
08005A92    cmp  r0, r1
08005A94    bgt  → 0x08005AC4            触发
08005ABC    subs r1, #0x28   (40)        ★★★ 基线 < 当前-40 ?
08005ABE    cmp  r0, r1
08005AC0    bge  → 不触发
08005AC4    ; —— 触发 ——
08005AD0      strh [ctx+0x15E6] = [ctx+0x1688]   记录触发时的快照索引
08005ADA      strb [ctx+0x15E4] = 1              ★ 事件标志
```

**⇒ 判据归纳：**

$$\boxed{\ |\,基线 - 当前\,| > 40 \ \Rightarrow\ \text{触发}\ }$$

**★ 注意这是「相对基线的变化量」，不是「绝对压力必须很大」—— 这个语义决定了后面的可改性（见 §六）。**

### 第 4 层 · 确认事件 + 采集一段

```asm
08005AEC  r0 = [ctx+0x15E4] ; cmp #1
08005AF6  bne → 0x08005B10
08005AFA  [ctx+0x168C] = 0
08005B06  [ctx+0x168A] = 1               ★★ 进入「采集中」

08005B12  r0 = [ctx+0x168A] ; beq → 返回
08005B28  k = 0 … cmp #3
08005B46    r3 = 0x578 (1400)            ★ 每通道记录块步长
08005B4A    r1 = ctx + k*0x578 + [ctx+0x15E2]*2
08005B58    strh [r1 + 0x28]             存 4 通道值

; —— ★★★ 采满 19 次才输出 ——
08005B68  [ctx+0x15E2]++                 环形索引自增
08005B74  r0 = [ctx+0x15E2] + 0x13 (19)
08005B7E  cmp r0, [sp,#6]（输入计数值） ; bne → 返回
08005B8C  r0 = [ctx+0x15E6]（触发时的快照索引）
08005B8C  bl 0x08005E38                  ★★★ 输出
08005B92  [ctx+0x15E2] = 0               清索引
08005BA0  [ctx+0x15E4] = 0               清事件标志
08005BA8  [ctx+0x168A] = 0               清采集中
08005BAE  [ctx] = 1                      设结果
```

**⇒ 所以震动不是「一次触发」，而是：**检测到压力突变 → 采集 19 个时间点 → 才输出**（相当于一个去抖/成形窗口）。**

### 第 5 层 · 打包成上报数据包

```asm
08005E38  参数 = 快照索引
08005E46  外层 i = 0 … 0x13 (20 个时间点)
08005E5E    r1 = 0xCCCCCCCD
08005E6A    r1 >>= 4
08005E6C    r1 += r1*4
08005E70    r0 -= r1<<2                  ★ 除法魔数 ⇒ index % 10
08005E74    [sp,#6] = 取模结果           10 槽环形缓冲
08005E8A    r0 = 0x2000495E + j*0x28     ★ ctx 基址（项目记录一致）
08005EA6      取环形缓冲[j*0x28 + index*2 + 0x15E8]
08005EB0    r1 = 0x2000495E + j*0x578
08005EBC    strh [r1 + i*2 + 2]          展开到结果块
```

**★ 魔数验证：`0xCCCCCCCD >> 35` 的标准除法是 ÷10 ⇒ `index = (index+1) % 10`。**
> **⇒ 输出 = 4 通道 × 20 时间点 = 80 个 16 位压力历史值（打包后经 `0x080110BC` 上报）。**

```asm
0800D748  r0 = 0x40000000
0800D750  bl 0x08010BF0                  初始化
0800D766  bl 0x080105AC
0800D76E  r2 = 0x2BC (700)               ★ CCR = 700（35% 占空比）
0800D772  bl 0x08010908                  ★★ 写 CCR

0800D78E  bl 0x08010D80
0800D794  bl 0x08010598
0800D79A  bl 0x08010BC8
0800D79E  r1 = [结构]
0800D7A2  orr r0, r0, #1 ; strb r0, [r1] ★ 标记「已启动」(bit0)
```

### ★★★ 一个关键的**否定性**发现

**写 CCR 的 HAL 函数 `0x08010908`，全固件**只有一个调用者**：**

```
L12611: 0800D772  bl #0x8010908      ← 就是上面那个「配置并启动」函数
```

> ## **⇒ 占空比只在上电初始化时设一次（0x2BC = 35%），运行期**从不修改 CCR**。**
> **⇒ 且 `0x0800FD64` 那串触觉模块初始化序列**没有任何 `bl` 调用者**（它调用的 `0x0800A430`/`0x0800E074`/`0x0800E798` 也都只被它调用一次）⇒ **它是经启动表进入的上电初始化**。
> **⇒ 而 TIM2 的结构（`0x200040B8`）在运行期**完全不被访问**（全固件只有 init 里的 2 处）。**

### 第 7 层 · 时序基础

```asm
; —— SysTick 忙等延时 ——
0800F0B8  muls r2, r0, r2
0800F0BE  r1 = 0xE000E010                ★ Cortex-M SysTick 寄存器
0800F0CE  orr r2, r2, #1
0800F0D2  str r2, [r1]                   使能 SysTick
0800F0D4  ; 忙等直到 [0x200040B4] == 0
0800F0E8  bic r0, r0, #1
0800F0EC  str r0, [r1]                   关闭 SysTick

; —— SysTick ISR（递减忙等计数器）——
0800CE7C  r0 = 0x200040B4
0800CE84  r1 = [r0] ; cmp #0 ; bxeq lr
0800CE8E  r1 -= 1
0800CE90  str r1, [r0]

; —— TIM3 侧计数器 ——
0800D6F4  r1 = 0x200040D0
0800D6FC  r0 = [r1, #0x10] ; r0 += 1 ; str r0, [r1, #0x10]

```

**★ 读该计数的两处：`0x080076B4`、`0x08007900`（在 I2C 从机命令处理里，给事件打时间戳）。**

## 二、★★★ 关键：灵敏度就是一个立即数

| 位置 | 指令 | 作用 |
|---|---|---|
| **`0x08005A90`** | `adds r1, #0x28` | 当前值 + 40 |
| **`0x08005ABC`** | `subs r1, #0x28` | 当前值 − 40 |

**⇒ 触发条件 = `基线 > 当前+40` **或** `基线 < 当前−40` = `|基线 − 当前| > 40`。**

**★ 把 `0x28` 改小 ⇒ 判据变敏感 ⇒ 极小的压力波动就触发。**

**而这正好对应「滑动时有震动反馈」：**
- 手指在板子上滑动时，压力**天然在波动**
- 判据变敏感后，滑动过程会**反复触发**
- **⇒ 滑动过程中就会反复震动** —— 而**完全不需要**主机触发、描述符、欺骗 Windows

## 四、❗ 未解出的一环（诚实标注）

**「运行期到底在哪里开关马达」** —— 我没有定位到。已知事实形成一个小矛盾：

| 已知 | 推论 |
|---|---|
| CCR 只在 init 设一次，运行期不改 | 占空比固定 |
| TIM2 结构运行期不被访问 | 没有运行期改频率/占空比的路径 |
| PWM 在上电初始化时启动 | **按理马达应该一直震** |
| **但实际只在点击时震** | **⇒ 一定存在一处我未找到的门控** |

**三种可能（按可能性）：**
1. **GPIO 门控驱动级使能**（PWM 常开，驱动级 EN 引脚开关）—— 最可能
2. 我漏了某处运行期对定时器输出使能位（CCER）的操作
3. 马达/驱动级有硬件级门控（如 RC 耦合 + 频率条件）

**★ 定位它的方法（下一步，本轮已把范围收窄）：**

**① GPIO 写 HAL 已识别：`0x0800F7FC`（引脚置位/复位，被大量调用）与 `0x0800F81C`（GPIO 初始化）。**

**② 外设访问指纹（`movw #imm + movt #0x4001` 配对统计）：**

| 外设 | 出现次数 |
|---|---|
| **GPIOA（0x40010800）** | **28** ← 最多，PA3 门铃 / PA9-PA10 USART1 都在这里 |
| 0x40013400 | 7 |
| 0x40012C00 | 7 |
| USART1/SPI2 区（0x40013800） | 6 |
| ADC1（0x40012400） | 3 |
| **GPIOB（0x40010C00）** | **1** |
| 其他（0x40015000 / 0x40012404 / 0x4001244C / 0x40014C00 / 0x40013804 / 0x40015400） | 各 1 |

**③ `0x0800F7FC` 的调用点分布（GPIO 写的可能位置）：**
`0x0800654C` · `0x080065A2` · `0x080065EE` · `0x08006620` · **`0x08007674`（I2C 从机命令处理，已确认）** · `0x08007BFC` · `0x08007CB0` · `0x08007D64` · `0x08007F08` · `0x0800888C` · `0x080099E4` · `0x08009A70` · `0x08009B84` · `0x08009C8E` · `0x08009CCA` · `0x0800A046` …

**★ 关键观察：这些调用点**没有一个落在触觉模块区（`0x0800D000`–`0x0800F000`）** ⇒ 说明马达门控**不在触觉模块内部**做 GPIO，而可能在 `0x080065xx` / `0x08007Bxx–0x08007Fxx` / `0x08009xxx` 这几处。**

**★ 下一步最有效的一步：把 `0x0800F7FC` 的每个调用点按「哪个 GPIO 口 + 哪个引脚 + 置位还是复位」解出来，找出"驱动级使能"那一个引脚。**
**★ 或直接夹 I2C1 逻辑分析仪，看点击瞬间哪个引脚动。**

**★ 补充：GPIOA 占 28 次、GPIOB 只 1 次 ⇒ 马达相关引脚几乎肯定在 **GPIOA** 上。**

## 五、对「滑动震动」目标的意义

| 路 | 状态 |
|---|---|
| 主机标准 HID 触发（`0x0E/0x21`） | ❌ 描述符无该 usage（`Col02 Out=0`） |
| 厂商私有通道触发（Col04，39 条命令） | ❌ 全表覆盖 + 定时器引用闭包 ⇒ 无马达入口 |
| 欺骗 Windows（改描述符） | ❌ 描述符只是声明，固件无对应处理 ⇒ 不震 |
| **★ 改 TF100A 灵敏度（本节）** | ✅ **判据立即数已定位，改 2 字节即可让「滑动时反复触发」** |

**⇒ 这是目前**唯一一条**有明确字节级目标、且直接对应「滑动时有震动」的路。**

## 七、方法学（本轮新增，可复用）

1. **用已知实测参数反查配置点**：用项目已实测的 `PSC=119 / ARR=1999 / CCR=0x2BC` 直接搜立即数，比顺着调用图找快得多。
2. **除法魔数识别法**：`0x1B4E81B5 >> 5` = ÷300、`0xCCCCCCCD >> 35` = ÷10 —— 一看到 `umull` + 魔数就能反推除数。
3. **「只有一个调用者」判据**：一个 HAL 写寄存器函数若全固件只有 1 个调用者，就能确定该寄存器**只在那一处被写**（本轮据此判定 CCR 固定）。
4. **「无 bl 调用者」= 启动表进入**：一段函数序列若无人 `bl`，且它调用的函数也都只被它调用 ⇒ 它是**上电初始化**，不是运行期路径。
5. **地址空间指纹**：`I2C1=0x40005400`、`GPIOA=0x40010800`、`ADC1=0x40012400`、`TIM2=0x40000000`、`TIM3=0x40000400` 全部吻合 STM32F1 ⇒ 可据此判断某个 `movt #0x4001` 到底指向哪个外设。

## 八、一句话总结

> **TF100A 的震动判据是「4 路 ADC 压力各自 |基线 − 当前| > 40」，而 `40` 就是 `0x08005A90` / `0x08005ABC` 两处的立即数 `0x28`。
> 把它改小，滑动时手指压力的天然波动就会反复跨过判据 ⇒ 在**不触碰主机协议、不欺骗 Windows** 的前提下做出「滑动时有震动反馈」。
> PWM 侧（TIM2 200 Hz / TIM3 400 Hz）在上电初始化时启动、CCR 运行期从不改变 —— 因此运行期「开关马达」的确切门控**仍是未解项**，是下一步要定位的目标。**
