【判读】
  0x229D4-0x229DC  movs r0,#0 ; bl 0x23A84 ; strh r0,[sp,#6]
     ⇒ 调用 0x23A84 取回一个 16 位值
  0x229E4  ldrh r0,[sp,#6]
  0x229E8  movw r1,#0xA55A
  0x229EC  cmp r0,r1 ; bne 0x229F4
     ⇒ ★ 如果读回值 == 0xA55A 则**跳过**写；否则写 (2, 0x0F9F)
  0x229F4  movs r0,#2 ; movw r1,#0x0F9F ; bl 0x1D6F0
     ⇒ 参数1 = 命令/类型 = 2；参数2 = 16 位数据 = 0x0F9F
